import argparse
import json
import os
import sys
import threading
import urllib.request
from pathlib import Path

BASE = "https://inference.poolside.ai/v1/chat/completions"
DEFAULT_MODEL = os.environ.get("POOLSIDE_MODEL", "poolside/laguna-s-2.1")


def call(key, model, system, task, timeout):
    body = {
        "model": model,
        "messages": [
            {"role": "system", "content": system},
            {"role": "user", "content": task},
        ],
        "temperature": 0.2,
        "max_tokens": 16384,
    }
    req = urllib.request.Request(
        BASE,
        data=json.dumps(body).encode(),
        headers={
            "Content-Type": "application/json",
            "Authorization": "Bearer " + key,
        },
        method="POST",
    )
    with urllib.request.urlopen(req, timeout=timeout) as r:
        data = json.load(r)
    return data["choices"][0]["message"].get("content", "").strip(), data.get("usage", {})


def main():
    ap = argparse.ArgumentParser(description="Send a code task to N parallel Poolside agents")
    ap.add_argument("--task", default=None, help="code writing task (inline text)")
    ap.add_argument("--task-file", default=None, help="path to a file holding the task text")
    ap.add_argument("--agents", type=int, choices=[1, 2, 3], required=True)
    ap.add_argument("--out", required=True, help="directory for agent outputs")
    ap.add_argument("--model", default=DEFAULT_MODEL)
    ap.add_argument(
        "--system",
        default=(
            "You are a senior software engineer. Write complete, correct, "
            "production-ready code for the requested task. Return the full "
            "implementation as ready-to-save files; use fenced code blocks with "
            "the filename in the opening fence (```path/to/file.py). No "
            "explanation outside the code unless asked."
        ),
    )
    ap.add_argument("--timeout", type=float, default=300)
    args = ap.parse_args()

    task = args.task
    if not task and args.task_file:
        task = Path(args.task_file).read_text(encoding="utf-8")
    if not task:
        ap.error("provide --task or --task-file")

    out = Path(args.out)
    out.mkdir(parents=True, exist_ok=True)

    keys = [os.environ.get("POOLSIDE_API_KEY_%d" % i) for i in range(1, args.agents + 1)]
    if any(not k for k in keys):
        print("error: missing POOLSIDE_API_KEY_1..%d in environment" % args.agents)
        sys.exit(2)

    results = {}

    def worker(i, key):
        try:
            content, usage = call(key, args.model, args.system, task, args.timeout)
            (out / ("agent_%d.md" % i)).write_text(content, encoding="utf-8")
            results[i] = ("OK", usage)
        except Exception as e:
            (out / ("agent_%d.error.txt" % i)).write_text(str(e), encoding="utf-8")
            results[i] = ("ERROR", str(e))

    threads = [
        threading.Thread(target=worker, args=(i, k)) for i, k in enumerate(keys, 1)
    ]
    for t in threads:
        t.start()
    for t in threads:
        t.join()

    ok = sum(1 for s, _ in results.values() if s == "OK")
    for i in sorted(results):
        print("agent_%d: %s %s" % (i, results[i][0], results[i][1]))
    sys.exit(0 if ok == len(keys) else 1)


if __name__ == "__main__":
    main()
